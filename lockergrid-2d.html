-- 1. Create KelownaInventory table
CREATE TABLE IF NOT EXISTS public."KelownaInventory" (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tag_id varchar(8) NOT NULL,
  part_name text NOT NULL,
  part_description text DEFAULT '',
  width_in numeric NOT NULL DEFAULT 18,
  height_in numeric NOT NULL DEFAULT 12,
  depth_in numeric NOT NULL DEFAULT 18,
  pos_x numeric NOT NULL DEFAULT 0,
  pos_y numeric NOT NULL DEFAULT 0,
  pos_z numeric NOT NULL DEFAULT 0,
  color text DEFAULT '#38bdf8',
  updated_at timestamptz DEFAULT now()
);

-- 2. Create KelownaInventoryLogs table for technician & 3rd-party tracking
CREATE TABLE IF NOT EXISTS public."KelownaInventoryLogs" (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  timestamp_label text NOT NULL,
  tech_name text NOT NULL,
  action_type text NOT NULL,
  tag_id varchar(8) DEFAULT '--',
  part_name text NOT NULL,
  part_request_number text DEFAULT '',
  note text DEFAULT '',
  created_at timestamptz DEFAULT now()
);

ALTER TABLE public."KelownaInventoryLogs"
ADD COLUMN IF NOT EXISTS part_request_number text DEFAULT '';

ALTER TABLE public."KelownaInventory" ENABLE ROW LEVEL SECURITY;
ALTER TABLE public."KelownaInventoryLogs" ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Allow full access to KelownaInventory" ON public."KelownaInventory";
CREATE POLICY "Allow full access to KelownaInventory"
ON public."KelownaInventory" FOR ALL USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS "Allow full access to KelownaInventoryLogs" ON public."KelownaInventoryLogs";
CREATE POLICY "Allow full access to KelownaInventoryLogs"
ON public."KelownaInventoryLogs" FOR ALL USING (true) WITH CHECK (true);

CREATE TABLE IF NOT EXISTS public."KelownaInventoryBackups" (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  trigger_reason text NOT NULL DEFAULT 'page-load',
  box_count integer NOT NULL DEFAULT 0,
  snapshot_json jsonb NOT NULL,
  created_at timestamptz DEFAULT now()
);
